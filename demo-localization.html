Code Examples

// Basic usage
import { localization } from './annie';

// Format numbers
localization.formatNumber(1234.56);
localization.formatCurrency(1234.56);
localization.formatPercent(0.1234);

// Format dates
localization.formatDate(new Date());
localization.formatTime(new Date());
localization.formatRelativeTime(new Date(Date.now() - 3600000));

// Message interpolation
localization.interpolate("Hello {{name}}, you have {{count | number}} messages", {
    name: "John",
    count: 5
});

// Typography helpers
localization.getTextDirection(); // 'ltr' or 'rtl'
localization.isRightToLeft(); // boolean
localization.applyTextDirection(element); // applies direction to DOM element
