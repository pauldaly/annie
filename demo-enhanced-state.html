🛠️ Code Examples

Basic Usage

// Create enhanced state manager
import { createEnhancedStateManager, createAction, createSelector } from './annie';

const stateManager = createEnhancedStateManager(logger, {
    enableActions: true,
    enableEffects: true,
    enableSelectors: true,
    strictMode: false
});

// Define actions
const increment = createAction<number>('INCREMENT');
const decrement = createAction<number>('DECREMENT');

// Register reducer
stateManager.registerReducer('counter', (state = 0, action) => {
    switch (action.type) {
        case 'INCREMENT':
            return state + (action.payload || 1);
        case 'DECREMENT':
            return state - (action.payload || 1);
        default:
            return state;
    }
});

// Dispatch actions
stateManager.dispatch(increment(5));
stateManager.dispatch(decrement(2));

// Use selectors
const doubleCounter = createSelector(
    (state) => state.counter * 2
);

stateManager.registerSelector('doubleCounter', doubleCounter);
const result = stateManager.select('doubleCounter');
