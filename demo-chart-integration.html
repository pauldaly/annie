1. Legacy Code Preservation

// Your existing createDashboard function (minimal changes) function createDashboard(id, data, drillid, filter, title) { // Existing logic preserved exactly as-is var svg = d3.selectAll("#" + id).append("svg") .attr("width", 450).attr("height", 250); svg.append("g").attr("id", id + "pie"); // Keep using gradPie if you have it if (window.gradPie) { gradPie.draw(id + "pie", piedata(data), 125, 130, 100); } // Your existing legend/title logic createLegend(svg, data, title); } // Annie wrapper - converts attributes to your function calls class DashboardRenderer { render(element, config, data) { createDashboard( element.id, this.convertData(data), config.drillid, config.filter, config.title ); } }

2. Annie Integration Layer

// Annie chart processor class AnnieChartProcessor { processChart(element) { // 1. Parse data attributes (with JsonLogic support) const config = this.parseAttributes(element); // 2. Get data from Annie DataStore const data = annie.dataset(config.source); // 3. Call your existing chart function const renderer = this.getRenderer(config.type); renderer.render(element, config, data); // 4. Setup observers for real-time updates this.setupObservers(element, config); } parseAttributes(element) { // Handle JsonLogic expressions in attributes const context = this.getContext(); // user, device, theme, etc. for (let attr of element.attributes) { if (this.isJsonLogic(attr.value)) { attr.value = jsonLogic.apply(JSON.parse(attr.value), context); } } } }
