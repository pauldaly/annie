3. Advanced List Rendering with Item Templates

data-observe='{ "datasource": "products", "type": "list", "templates": { "item": "<div class=\"product-item\"><span>{{this.name}}</span><span class=\"price\">{{this.price | currency:USD}}</span></div>", "empty": "<div class=\"empty\">No products available</div>" } }'

4. Loop Templates with Context Variables

data-observe='{ "datasource": "users", "type": "template", "template": "{{#each users}}<div class=\"user-card\">User {{@index}}: {{this.name}} ({{#if @first}}First{{/if}}{{#if @last}}Last{{/if}})</div>{{/each}}" }'
Loading users...

5. State-Aware Templates (Loading/Error/Empty)

data-observe='{ "datasource": "apiData", "type": "html", "templates": { "loading": "<div class=\"loading\">⏳ Loading data...</div>", "error": "<div class=\"error\">❌ Error: {{error.message}}</div>", "empty": "<div class=\"empty\">📭 No data available</div>" }, "template": "<div>Data loaded: {{count}} items</div>" }'
